Function validation(username,password)
    If username = "" or password = "" then
        Output "Username or password cannot be empty"

    Endif
    Return TRUE
EndFunction

Function check_exist_user(username)
    Open "account.txt" For reading
    If file does not exist then
        Return FALSE
    Endif

    For each line in file
        stored_username # the first value in line
        If stored_username = username then
            Return TRUE
        Endif
    EndFor

    Return FALSE
EndFunction


Procedure store_acc(username, password)
    hashed_password = hash_password(password)
    Open "account.txt" For appending
        Write username , hashed_password
EndProcedure


Procedure register()
    username # value from entry_username
    If check_exist_user(username) then
        Output "Account already exist"
        Return
    Endif
    password # value from entry_password
    If validation(username,password) = FALSE then
        Return
    Endif
    create UUID # unique identifier for each player
    If UUID already exist then
        create new_UUID
    Endif
    store_acc(username, password,UUID) 
    create_savings()
    Output "Account created successfully"
EndProcedure


Function login()
    username # value from entry_username
    password # value from entry_password
    hashed_password = hash_password(password)

    Open "account.txt" For reading
    If file does not exist then
        Output "Invalid username or password."
        Return FALSE
    Endif

    For each line in file
        stored_username # the first value in line
        stored_hashed_password # the second value in line
        If stored_username = username AND stored_hashed_password = hashed_password then
            Output "Login successful."
            Return TRUE
        Endif
    EndFor

    Output "Invalid username or password."
    Return FALSE
EndFunction

Procedure load_map(screen,map_data,world_offset)
    For layers in map_data 
        For tiles in layers 
            If tile exist then
                world_offset_x , world_offset_y = get_world_offset()
                x_pixel = tile[0] * pixel_size + world_offset_x
                y_pixel = tile[1] * pixel_size + world_offset_y
                Add tile to screen
            EndIf
        EndFor
    EndFor
EndProcedure


Function get_world_offset(player, screen_width, screen_height)
    world_offset_x = player.x - (screen_width/2)
    world_offset_y = player.y - (screen_height/2)
    Return world_offset_x, world_offset_y
EndFunction
