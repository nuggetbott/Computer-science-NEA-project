Function hash_password(password)
    hashed # SHA256 hash of password
    Return hashed
EndFunction


Function check_exist_user(username)
    Open "account.txt" For reading
    If file does not exist then
        Return FALSE
    Endif

    For each line in file
        stored_username # the first value in line
        If stored_username = username then
            Return TRUE
        Endif
    EndFor

    Return FALSE
EndFunction


Procedure store_acc(username, password)
    hashed_password = hash_password(password)
    Open "account.txt" For appending
        Write username , hashed_password
EndProcedure


Function register()
    Input username
    If check_exist_user(username) then
        Output "Account already exist"
        Return
    Endif
    Input password
    store_acc(username, password)
    Output "Account created successfully"
EndFunction


Function login()
    Input username
    Input password
    hashed_password = hash_password(password)

    Open "account.txt" For reading
    If file does not exist then
        Output "Invalid username or password."
        Return FALSE
    Endif

    For each line in file
        stored_username # the first value in line
        stored_hashed_password # the second value in line
        If stored_username = username AND stored_hashed_password = hashed_password then
            Output "Login successful."
            Return TRUE
        Endif
    EndFor

    Output "Invalid username or password."
    Return FALSE
EndFunction


Input option # "register" or "login"
If option = "register" then
    register()
Elseif option = "login" then
    login()
Else
    Output "Invalid option"
Endif
