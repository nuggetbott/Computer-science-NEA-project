# Psuedocode for login and register

Function hash_password(password)
    hashed # SHA256 hash of password
    Return hashed
EndFunction

Function validation(username,password)
    If username = "" or password = "" then
        Output "Username or password cannot be empty"
        Return FALSE
    Endif
    Return TRUE
EndFunction

#Validate username & password
If username = "" or password = "" then
    Output "Error,  username or password cannt be empty"
Endif
If  " " in username  or " " in password then
    Output "Error, username or password cannot contain spaces"
Endif




Function check_exist_user(username)
    Open "account.txt" For reading
    If file does not exist then
        Return FALSE
    Endif

    For each line in file
        stored_username # the first value in line
        If stored_username = username then
            Return TRUE
        Endif
    EndFor

    Return FALSE
EndFunction


Procedure store_acc(username, password)
    hashed_password = hash_password(password)
    Open "account.txt" For appending
        Write username , hashed_password
EndProcedure


Function register()
    username # value from entry_username
    If check_exist_user(username) then
        Output "Account already exist"
        Return
    Endif
    password # value from entry_password
    If validation(username,password) = FALSE then
        Return
    Endif
    create UUID # unique identifier for each player
    If UUID already exist then
        create new_UUID
    Endif
    store_acc(username, password,UUID) 
    create_savings()
    Output "Account created successfully"
EndFunction


Function login()
    username # value from entry_username
    password # value from entry_password
    hashed_password = hash_password(password)

    Open "account.txt" For reading
    If file does not exist then
        Output "Invalid username or password."
        Return FALSE
    Endif

    For each line in file
        stored_username # the first value in line
        stored_hashed_password # the second value in line
        If stored_username = username AND stored_hashed_password = hashed_password then
            Output "Login successful."
            Return TRUE
        Endif
    EndFor

    Output "Invalid username or password."
    Return FALSE
EndFunction

#Saving/load system
save_file = "save_file.json"
temp_save_file = "save_file.tmp"

Function load_game():
    If save_file does not exist then
        Return not exist
    Endif
    Open "save_file.son" for reading 
        Read JSON data 
    Convert JSON data into game data
    Close "save_file.json"
EndFunction

Function create_savings():
    If save_file does not exist then
        Create "save_file.json"
        Create default account
        Return "Savings created"
    Return "file already exists"
EndFunction

Function save_game():
    Convert game_data into JSON
    Open temp_save_file for writing
        Write game_data
    Close temp_save_file

    If temp_save_file write was successful then
        Replace save_file with temp_save_file
        Return "Game saved successfully"
    else:
        Return "Error-Game not saved"
